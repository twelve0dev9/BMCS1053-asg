SET DEFINE ON
SET LINESIZE 300
SET PAGESIZE 100
SET VERIFY OFF

PROMPT Choose Speaker:

-- list all speakers
SELECT Speaker_ID || '. ' || Name AS "Speakers List"
FROM Speakers
ORDER BY Speaker_ID;

-- Accept user input
ACCEPT CHOICE NUMBER PROMPT 'Choose (0 for All speaker) : '

-- Define column formatting
COLUMN Speaker_Name FORMAT A25 HEADING 'Speaker Name'
COLUMN Phone_Number FORMAT A15 HEADING 'Phone Number'
COLUMN Email        FORMAT A30 HEADING 'Email'
COLUMN Total_Events FORMAT 9999 HEADING 'Total Events'

-- Query to display events count for a single speaker,
-- or for all speakers if option 0 is selected
SELECT s.Name AS Speaker_Name,
       s.Phone_Number,
       s.Email,
       COUNT(e.Event_ID) AS Total_Events
FROM Speakers s
JOIN Events_Participation ep ON s.Speaker_ID = ep.Speaker_ID
JOIN Events e ON ep.Event_ID = e.Event_ID
WHERE &CHOICE = 0 OR s.Speaker_ID = &CHOICE
GROUP BY s.Speaker_ID, s.Name, s.Phone_Number, s.Email
ORDER BY s.Speaker_ID;
